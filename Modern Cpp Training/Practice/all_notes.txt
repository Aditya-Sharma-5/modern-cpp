std::optional<std::list<int> > FIndOddNUmber(std::list<int>& data){ 
    if(data.empty()){
        throw std::runtime_error("Empty Data ");
    }
    std::list<int> result;
    for(int val: data){
        if(val % 2 != 0){
            result.push_back(val);
        }
    }
    if(result.empty()){
        return std::nullopt;
    }
    return result;
}

int main(){
    std::list<int> l1{1 , 2, 3, 4, 5};
    std::list<int> l2{6 , 2, 4};
    std::optional<std::list<int> >  result = FIndOddNUmber(l2);
     
    if(result.has_value()){
        std::list<int>valid_data = result.value();
        for(int val:valid_data){
            std::cout<<val<<" ";
        }
        std::cout<<"\n";
    }else{
        std::cerr<<"Function returned null!!";
    }
    return 0 ;
}

using Vtype = std::variant<int , std::string>;
using Container = std::array<Vtype , 3>;

void performOperation(Container& v){
    for(Vtype v:v)
    {
        if(std::holds_alternative<int>(v))
        {
            int value = std::get<0>(v);
            std::cout<<"Square of "<<value<<" is : "<<value*value<<"\n";
        }
        else
        {
            std::string& data = std::get<1>(v);
            std::cout<<"Length is : "<<data.length()<<"\n";
        }
    }
}

void CommonOPeration(Container& v){
    for(Vtype v: v)
    {
        std::visit([](auto&& val){std::cout<<val<<"\n";},v);
    }
}
int main()
    Container vr;
    vr[0] = 10;
    vr[1] = "Aadi";
    vr[2] = "Be Positive";

    performOperation(vr);
    CommonOPeration(vr);

int fact(std::future<int>& ft){
    int num = ft.get();   // We cannot proceed from this line before pr.set_value();
    if(num<0){
        throw std::runtime_error{"Negative Number"};
    }
    if(num == 0 || num ==1){
        return 1 ;
    }
    int ans = 1; 
    for(int i = 2; i<=num ; i++){
        ans *=i;
    }
    return  ans ;
}

main :  
    std::promise<int>pr;
    std::future<int>ft = pr.get_future();
    std::future<int>ans = std::async(&fact , std::ref(ft)); 
    int number = takeInput();   
    pr.set_value(number);    // this will notify the functions 
    std::cout<<ans.get()<<"\n";
