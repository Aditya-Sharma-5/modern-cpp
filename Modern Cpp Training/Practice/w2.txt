void Adapter(const std::array<int , 5>& data , std::function<int(int)>fn){
    for(const int val:data){
        std::cout<<fn(val)<<"\n";
    }
}

//transform_if
void Adaptor(const std::array<int , 5>& arr , std::function<bool(int)>fn)
{
    for(const int i:arr){
        if(fn(i)){
            std::cout<<i<<" ";
        }
    }
    std::cout<<"\n";
}

//std::transform
void Adaptor1(const std::array<int , 5>& arr , std::function<const int(const int)>fn){

    for(const int val:arr){
        std::cout<<val<<"\n";
    }
    std::cout<<"\n";
}

int main(){
    std::array<int , 5>arr{10 , 20 , 30 , 40 , 50};

    Adaptor(arr , [](const int num){return (num%4 == 0);});

    Adaptor(arr , [](const int num){return (num >10);});

    Adaptor1(arr , [&](int num){ return  num*num;});
}
